import bpy

W, L, H, T, HOLE = 40, 60, 50, 6, 6.5

def box(name, sx, sy, sz, x=0.0, y=0.0, z=0.0):
    bpy.ops.mesh.primitive_cube_add(size=1, location=(x + sx / 2, y + sy / 2, z + sz / 2))
    obj = bpy.context.active_object
    obj.scale = (sx, sy, sz)
    bpy.ops.object.transform_apply(scale=True)
    obj.name = name
    return obj

def cyl(name, r, h, location, rotation=(0, 0, 0)):
    bpy.ops.mesh.primitive_cylinder_add(vertices=32, radius=r, depth=h, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def boolean(target, tool, op):
    mod = target.modifiers.new(name=op, type='BOOLEAN')
    mod.operation = op
    mod.object = tool
    mod.solver = 'EXACT'
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(tool, do_unlink=True)

bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete()

bracket = box("bracket", L, W, T)
boolean(bracket, box("wall", T, W, H), 'UNION')

# gusset: a triangular prism from a mesh
gl, gh, gt = L * 0.6, H * 0.6, 4
verts = [(T, W / 2 - gt / 2, T), (T + gl, W / 2 - gt / 2, T), (T, W / 2 - gt / 2, T + gh),
         (T, W / 2 + gt / 2, T), (T + gl, W / 2 + gt / 2, T), (T, W / 2 + gt / 2, T + gh)]
faces = [(0, 1, 2), (3, 5, 4), (0, 3, 4, 1), (1, 4, 5, 2), (2, 5, 3, 0)]
mesh = bpy.data.meshes.new("gusset")
mesh.from_pydata(verts, [], faces)
gusset = bpy.data.objects.new("gusset", mesh)
bpy.context.collection.objects.link(gusset)
boolean(bracket, gusset, 'UNION')

for y in (W / 4, 3 * W / 4):
    boolean(bracket, cyl("foot_hole", HOLE / 2, T * 3, (L * 0.7, y, T / 2)), 'DIFFERENCE')
boolean(bracket, cyl("wall_hole", HOLE / 2, T * 3, (T / 2, W / 2, H * 0.75), rotation=(0, 1.5708, 0)), 'DIFFERENCE')

bpy.ops.export_scene.gltf(filepath="/tmp/bracket.glb", export_format='GLB')
